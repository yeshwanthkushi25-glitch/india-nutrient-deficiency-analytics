4. KEY DAX MEASURES IMPLEMENTED IN THE MODEL:
--------------------------------------------------------------------------------
1. Caloric_Adequacy_Pct = 
   DIVIDE(AVERAGE(Fact_State_Nutrition[Avg_Daily_Calorie_Intake_kcal]), 1734.7, 0) * 100

2. Protein_Deficit_Rate = 
   AVERAGE(Fact_State_Nutrition[Protein_Deficit_Pct])

3. Carb_Energy_Share = 
   AVERAGE(Fact_State_Nutrition[Carbohydrate_Energy_Share_Pct])

4. Hidden_Hunger_Severity_Index = 
   AVERAGE(Fact_State_Nutrition[Hidden_Hunger_Severity_Index])

5. Awareness_to_Deficiency_Ratio = 
   DIVIDE(AVERAGE(Fact_State_Nutrition[Nutritional_Literacy_Score]), 
          AVERAGE(Fact_State_Nutrition[Hidden_Hunger_Severity_Index]) + 0.001, 0)

6. Anemia_Prevalence_Rate = 
   AVERAGE(Fact_State_Nutrition[Iron_Deficiency_Anemia_Pct])

7. Severe_Vulnerability_Flag = 
   IF([Hidden_Hunger_Severity_Index] >= 75, "CRITICAL INTERVENTION", "MONITOR")
